' =============================================================================
'  PUNTO DE ENTRADA - asignado al boton en el Dashboard
' =============================================================================
Public Sub CargarYAnalizarCondicion()
    Dim calcMode    As XlCalculation
    Dim filePath    As String
    Dim nombreArch  As String
    
    With Application.FileDialog(msoFileDialogFilePicker)
        .Title = "Seleccionar exportacion SAP VK13"
        .Filters.Clear
        .Filters.Add "Archivos de texto", "*.txt;*.dat;*.prn;*.csv"
        .Filters.Add "Todos los archivos", "*.*"
        .AllowMultiSelect = False
        If .Show <> -1 Then Exit Sub
        filePath = .SelectedItems(1)
    End With
    
    nombreArch = Mid(filePath, InStrRev(filePath, "\") + 1)
    
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    calcMode = Application.Calculation
    Application.Calculation = xlCalculationManual
    
    On Error GoTo ErrorHandler
    
    Dim registros() As RegistroVK13
    Dim totalReg    As Long
    
    totalReg = ParsearArchivoSAP(filePath, registros)
    
    If totalReg = 0 Then
        MsgBox "No se encontraron registros MXN válidos en el archivo.", _
               vbExclamation, "CEMEX VK13 - Sin datos"
        GoTo Finalizar
    End If
    
    EscribirHojaDatos registros, totalReg
    ActualizarDashboard nombreArch
    
    MsgBox "Proceso completado con éxito." & vbCrLf & vbCrLf & _
           ChrW(10004) & " Registros analizados : " & totalReg, _
           vbInformation, "CEMEX | Control de Condiciones VK13"
    GoTo Finalizar

ErrorHandler:
    MsgBox "Error Principal " & Err.Number & ": " & Err.Description, vbCritical, "CEMEX VK13"
    
Finalizar:
    Application.Calculation = calcMode
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
End Sub

' =============================================================================
'  PARSER SAP VK13 - Maquina de estados con Failsafe
' =============================================================================
Public Function ParsearArchivoSAP(filePath As String, ByRef registros() As RegistroVK13) As Long
    Dim fso     As Object
    Dim ts      As Object
    Dim linea   As String
    Dim campos() As String
    Dim estado  As RegistroVK13
    Dim count   As Long
    Dim etiq    As String
    Dim limiteSeguridad As Long

    Set fso = CreateObject("Scripting.FileSystemObject")
    On Error Resume Next
    Set ts = fso.OpenTextFile(filePath, 1, False, 0)
    If Err.Number <> 0 Then
        Err.Clear
        Set ts = fso.OpenTextFile(filePath, 1, False, -1)
    End If
    On Error GoTo 0

    ReDim registros(0 To 5000)
    count = 0
    limiteSeguridad = 0
    InicializarEstado estado

    Do While Not ts.AtEndOfStream
        limiteSeguridad = limiteSeguridad + 1
        If limiteSeguridad > 300000 Then Exit Do

        linea = ts.ReadLine
        campos = Split(linea, vbTab)
        
        If UBound(campos) >= 0 Then
            etiq = Trim(LCase(QuitarTildes(campos(0))))
            
            If Left(etiq, 18) = "clase de condicion" Then
                estado.ClaseCondicion = ValorCampo(campos, 1)
                estado.LlaveTxt = UltimoValorCampo(campos, 2)
                
            ElseIf InStr(1, etiq, "organizacion") > 0 Then
                estado.EmisorID = ValorCampo(campos, 1)
                estado.EmisorDesc = UltimoValorCampo(campos, 2)
                
            ElseIf etiq = "centro" Then
                estado.Centro = ValorCampo(campos, 1)
                estado.CentroDesc = UltimoValorCampo(campos, 2)
                
            ElseIf etiq = "material" Then
                estado.Material = ValorCampo(campos, 1)
                estado.Denominacion = UltimoValorCampo(campos, 2)
                
            ElseIf Left(etiq, 18) = "periodo de validez" Then
                estado.ValidoDe = ExtraerFecha(campos, True)
                estado.FinValidez = ExtraerFecha(campos, False)
                
            ElseIf Left(etiq, 9) = "creado el" Then
                estado.Creado = ExtraerFecha(campos, True)
                
            ElseIf ContieneValor(campos, "MXN") Then
                estado.Importe = ParsearDouble(ExtraerImporte(campos))
                estado.ImporteValido = (ExtraerImporte(campos) <> "")
                estado.Moneda = "MXN"
                estado.UM = ExtraerUM(campos)
                AsignarModalidad estado
                
                If count > UBound(registros) - 1 Then
                    ReDim Preserve registros(0 To UBound(registros) + 2000)
                End If
                
                registros(count) = estado
                count = count + 1
                InicializarEstado estado
            End If
        End If
    Loop

    ts.Close
    Set ts = Nothing
    Set fso = Nothing
    ParsearArchivoSAP = count
End Function

Public Sub InicializarEstado(ByRef e As RegistroVK13)
    e.ClaseCondicion = ""
    e.LlaveTxt = ""
    e.Modalidad = ""
    e.ModalidadTxt = ""
    e.EmisorID = ""
    e.EmisorDesc = ""
    e.Centro = ""
    e.CentroDesc = ""
    e.Material = ""
    e.Denominacion = ""
    e.Importe = 0
    e.ImporteValido = False
    e.Moneda = ""
    e.UM = ""
    e.ValidoDe = ""
    e.FinValidez = ""
    e.Creado = ""
End Sub

Public Sub AsignarModalidad(ByRef e As RegistroVK13)
    Select Case e.EmisorID
        Case "7100"
            If UCase(e.ClaseCondicion) = "ZMPH" Then
                e.Modalidad = "ZMPH"
                e.ModalidadTxt = "FLETE"
            Else
                e.Modalidad = "ZMAH"
                e.ModalidadTxt = "MATERIAL"
            End If
        Case "7180"
            If UCase(e.ClaseCondicion) = "ZMP1" Then
                e.Modalidad = "ZMP1"
                e.ModalidadTxt = "FLETE"
            Else
                e.Modalidad = "ZMA6"
                e.ModalidadTxt = "MATERIAL"
            End If
        Case Else
            If e.ClaseCondicion <> "" Then
                e.Modalidad = e.ClaseCondicion
            Else
                e.Modalidad = "GENERAL"
            End If
            e.ModalidadTxt = "GENERAL"
    End Select
End Sub

Public Function ValorCampo(campos() As String, pos As Integer) As String
    Dim i As Integer
    Dim c As Integer
    c = 0
    For i = 1 To UBound(campos)
        If Trim(campos(i)) <> "" Then
            c = c + 1
            If c = pos Then
                ValorCampo = Trim(campos(i))
                Exit Function
            End If
        End If
    Next i
    ValorCampo = ""
End Function

Public Function UltimoValorCampo(campos() As String, startPos As Integer) As String
    Dim i As Integer
    Dim c As Integer
    Dim last As String
    c = 0
    last = ""
    For i = 1 To UBound(campos)
        If Trim(campos(i)) <> "" Then
            c = c + 1
            If c >= startPos Then
                last = Trim(campos(i))
            End If
        End If
    Next i
    UltimoValorCampo = last
End Function

Public Function ExtraerFecha(campos() As String, bFirst As Boolean) As String
    Dim i As Integer
    Dim v As String
    Dim primera As String
    Dim ultima As String
    primera = ""
    ultima = ""
    For i = 0 To UBound(campos)
        v = Trim(campos(i))
        If Len(v) = 10 Then
            If Mid(v, 3, 1) = "." And Mid(v, 6, 1) = "." Then
                If primera = "" Then
                    primera = v
                End If
                ultima = v
            End If
        End If
    Next i
    
    If bFirst Then
        ExtraerFecha = primera
    Else
        ExtraerFecha = ultima
    End If
End Function

Public Function ContieneValor(campos() As String, valor As String) As Boolean
    Dim i As Integer
    For i = 0 To UBound(campos)
        If UCase(Trim(campos(i))) = UCase(valor) Then
            ContieneValor = True
            Exit Function
        End If
    Next i
    ContieneValor = False
End Function

Public Function ExtraerImporte(campos() As String) As String
    Dim i As Integer
    Dim j As Integer
    Dim v As String
    For i = 0 To UBound(campos)
        If UCase(Trim(campos(i))) = "MXN" Then
            For j = i - 1 To 0 Step -1
                v = Trim(campos(j))
                If v <> "" Then
                    Dim vt As String
                    vt = Replace(Replace(Replace(v, ",", ""), ".", ""), " ", "")
                    If IsNumeric(vt) Then
                        ExtraerImporte = v
                        Exit Function
                    End If
                    vt = Replace(Replace(v, ",", "."), " ", "")
                    If IsNumeric(vt) Then
                        ExtraerImporte = v
                        Exit Function
                    End If
                End If
            Next j
            ExtraerImporte = ""
            Exit Function
        End If
    Next i
    ExtraerImporte = ""
End Function

Public Function ExtraerUM(campos() As String) As String
    Dim i As Integer
    Dim enc As Boolean
    Dim last As String
    enc = False
    last = ""
    For i = 0 To UBound(campos)
        If UCase(Trim(campos(i))) = "MXN" Then
            enc = True
        End If
        If enc And i > 0 Then
            Dim v2 As String
            v2 = Trim(campos(i))
            If v2 <> "" And UCase(v2) <> "MXN" And UCase(v2) <> "POR" Then
                last = v2
            End If
        End If
    Next i
    ExtraerUM = last
End Function

Public Function ParsearDouble(v As String) As Double
    If v = "" Then
        ParsearDouble = 0
        Exit Function
    End If
    
    Dim s As String
    s = Trim(v)
    
    If InStr(s, ",") > 0 And InStr(s, ".") > 0 Then
        If InStrRev(s, ".") < InStrRev(s, ",") Then
            s = Replace(s, ".", "")
            s = Replace(s, ",", ".")
        Else
            s = Replace(s, ",", "")
        End If
    ElseIf InStr(s, ",") > 0 Then
        s = Replace(s, ",", ".")
    End If
    
    s = Replace(s, " ", "")
    
    On Error Resume Next
    ParsearDouble = CDbl(s)
    If Err.Number <> 0 Then
        ParsearDouble = 0
        Err.Clear
    End If
    On Error GoTo 0
End Function

Public Function QuitarTildes(s As String) As String
    Dim r As String
    r = s
    r = Replace(r, Chr(225), "a")
    r = Replace(r, Chr(233), "e")
    r = Replace(r, Chr(237), "i")
    r = Replace(r, Chr(243), "o")
    r = Replace(r, Chr(250), "u")
    r = Replace(r, Chr(241), "n")
    r = Replace(r, Chr(193), "a")
    r = Replace(r, Chr(201), "e")
    r = Replace(r, Chr(205), "i")
    r = Replace(r, Chr(211), "o")
    r = Replace(r, Chr(218), "u")
    r = Replace(r, Chr(209), "n")
    QuitarTildes = r
End Function

' =============================================================================
Public Sub EscribirHojaDatos(registros() As RegistroVK13, totalReg As Long)
    Dim ws As Worksheet
    Dim i As Long
    Dim r As RegistroVK13
    
    Set ws = ObtenerOCrearHoja("Datos")
    ws.Cells.Clear
    
    With ws
        .Cells(1, 1) = "Clase de condici" & Chr(243) & "n"
        .Cells(1, 2) = "Llave (texto)"
        .Cells(1, 3) = "Modalidad"
        .Cells(1, 4) = "Modalidad Texto"
        .Cells(1, 5) = "Emisor ID"
        .Cells(1, 6) = "Emisor Desc"
        .Cells(1, 7) = "Centro"
        .Cells(1, 8) = "Centro Desc"
        .Cells(1, 9) = "Material"
        .Cells(1, 10) = "Denominaci" & Chr(243) & "n"
        .Cells(1, 11) = "Importe"
        .Cells(1, 12) = "Moneda"
        .Cells(1, 13) = "UM"
        .Cells(1, 14) = "Valido de"
        .Cells(1, 15) = "Fin validez"
        .Cells(1, 16) = "Creado"
    End With
    
    With ws.Range(ws.Cells(1, 1), ws.Cells(1, 16))
        .Interior.Color = CLR_NAVY
        .Font.Color = CLR_WHITE
        .Font.Bold = True
        .Font.Name = "Segoe UI"
        .Font.Size = 9
        .RowHeight = 20
    End With
    
    For i = 0 To totalReg - 1
        r = registros(i)
        With ws
            .Cells(i + 2, 1) = r.ClaseCondicion
            .Cells(i + 2, 2) = r.LlaveTxt
            .Cells(i + 2, 3) = r.Modalidad
            .Cells(i + 2, 4) = r.ModalidadTxt
            .Cells(i + 2, 5) = r.EmisorID
            .Cells(i + 2, 6) = r.EmisorDesc
            .Cells(i + 2, 7) = r.Centro
            .Cells(i + 2, 8) = r.CentroDesc
            .Cells(i + 2, 9) = r.Material
            .Cells(i + 2, 10) = r.Denominacion
            .Cells(i + 2, 11) = r.Importe
            .Cells(i + 2, 12) = r.Moneda
            .Cells(i + 2, 13) = r.UM
            .Cells(i + 2, 14) = r.ValidoDe
            .Cells(i + 2, 15) = r.FinValidez
            .Cells(i + 2, 16) = r.Creado
        End With
    Next i
    
    If totalReg > 0 Then
        ws.Range(ws.Cells(2, 11), ws.Cells(totalReg + 1, 11)).NumberFormat = "$#,##0.00"
    End If
    
    ws.Columns("A:P").AutoFit
End Sub

' =============================================================================
'  DASHBOARD PRINCIPAL (UI/UX EJECUTIVO CORPORATIVO)
' =============================================================================
Public Sub ActualizarDashboard(Optional nombreArchivo As String = "")
    Dim calcMode As XlCalculation
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    calcMode = Application.Calculation
    Application.Calculation = xlCalculationManual
    On Error GoTo ErrDash

    Dim wsDash As Worksheet
    Dim wsDatos As Worksheet
    Dim wsData As Worksheet
    
    Set wsDash = ObtenerOCrearHoja("Dashboard")
    Set wsDatos = ObtenerHoja("Datos")
    Set wsData = ObtenerOCrearHoja("_DashData")

    If wsDatos Is Nothing Then
        MsgBox "No existe la hoja Datos. Cargue primero un archivo SAP.", vbExclamation, "CEMEX VK13"
        GoTo FinDash
    End If

    wsDash.Activate
    ActiveWindow.DisplayGridlines = False
    wsData.Visible = xlSheetVeryHidden

    Dim lastRow As Long
    lastRow = wsDatos.Cells(wsDatos.Rows.count, 1).End(xlUp).Row

    Dim anomalias() As Long
    Dim totalAnom As Long
    totalAnom = ContarYClasificarAnomalias(wsDatos, lastRow, anomalias)

    ' Limpieza previa de celdas
    wsDash.Cells.Clear
    
    ' Limpieza segura de gráficos y formas (conservando o regenerando el botón)
    On Error Resume Next
    Dim co As ChartObject
    For Each co In wsDash.ChartObjects
        co.Delete
    Next co
    
    Dim sh As Shape
    For Each sh In wsDash.Shapes
        If sh.Name <> "BtnCargar" Then
            sh.Delete
        End If
    Next sh
    On Error GoTo ErrDash

    ' Configuración de Columnas
    wsDash.Columns("A").ColumnWidth = 3.5    ' Sidebar tipo App Web
    Dim ci As Integer
    For ci = 2 To 13
        wsDash.Columns(ci).ColumnWidth = 18.5
    Next ci
    wsDash.Columns("N").ColumnWidth = 2      ' Margen derecho
    
    ' Configuración de Filas
    wsDash.Rows(1).RowHeight = 6             ' Padding superior
    wsDash.Rows(2).RowHeight = 28            ' Hero Título
    wsDash.Rows(3).RowHeight = 18            ' Hero Subtítulo
    wsDash.Rows(4).RowHeight = 24            ' Caja de Resumen Dinámico
    wsDash.Rows(5).RowHeight = 8             ' Separador antes de KPIs
    wsDash.Rows(6).RowHeight = 20            ' KPIs Fila 1
    wsDash.Rows(7).RowHeight = 30            ' KPIs Fila 2 (Valores)
    wsDash.Rows(8).RowHeight = 18            ' KPIs Fila 3 (Subs)
    wsDash.Rows(9).RowHeight = 8             ' Separador después de KPIs
    wsDash.Rows(10).RowHeight = 18           ' Encabezado Sección Analítica
    
    Dim rChart As Long
    For rChart = 11 To 24
        wsDash.Rows(rChart).RowHeight = 14   ' Área de Gráficos (14 filas = 196 pt)
    Next rChart
    
    wsDash.Rows(25).RowHeight = 8            ' Separador de auditoría
    wsDash.Rows(26).RowHeight = 18           ' Encabezado Auditoría
    wsDash.Rows(27).RowHeight = 18           ' Headers Tabla

    ' --- 1. SIDEBAR LATERAL ESTILO APP WEB (COLUMNA A) ---
    wsDash.Columns("A").Interior.Color = CLR_SIDEBAR
    With wsDash.Columns("A").Borders(xlEdgeRight)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .Color = CLR_SIDEBAR_BOR
    End With

    ' --- 2. HERO BANNER ---
    wsDash.Range("B1:M3").Interior.Color = CLR_NAVY
    DibujarCeldaMerge wsDash, 2, 2, 2, 13, "CEMEX  |  Monitoreo y Control de Condiciones VK13", _
                       15, True, CLR_WHITE, xlLeft, xlCenter
                       
    Dim subTxt As String
    If nombreArchivo = "" Then
        subTxt = "  Fuente: Registros activos en SAP   |   Actualizado: " & Format(Now, "dd/mm/yyyy hh:mm")
    Else
        subTxt = "  Fuente: " & nombreArchivo & "   |   Actualizado: " & Format(Now, "dd/mm/yyyy hh:mm")
    End If
    DibujarCeldaMerge wsDash, 3, 2, 3, 13, subTxt, 9, False, CLR_CARD_BORDER, xlLeft, xlCenter

    ' --- 3. CÁLCULO DE MÉTRICAS GENERALES ---
    Dim totalReg As Long
    If lastRow > 1 Then
        totalReg = lastRow - 1
    Else
        totalReg = 0
    End If
    
    Dim impProm As Double
    Dim cnt7100 As Long
    Dim cnt7180 As Long
    Dim sumaImp As Double
    Dim cntImp As Long
    Dim i As Long
    
    cnt7100 = 0
    cnt7180 = 0
    sumaImp = 0
    cntImp = 0
    
    For i = 2 To lastRow
        Dim emK As String
        Dim impK As Variant
        emK = Trim(CStr(wsDatos.Cells(i, 5).Value))
        impK = wsDatos.Cells(i, 11).Value
        
        If emK = "7100" Then cnt7100 = cnt7100 + 1
        If emK = "7180" Then cnt7180 = cnt7180 + 1
        
        If IsNumeric(impK) Then
            If CDbl(impK) > 0 Then
                sumaImp = sumaImp + CDbl(impK)
                cntImp = cntImp + 1
            End If
        End If
    Next i
    
    If cntImp > 0 Then
        impProm = sumaImp / cntImp
    Else
        impProm = 0
    End If

    ' Preparar datos analíticos de soporte (_DashData)
    Dim topCentroNom As String
    Dim topCentroProm As Double
    Dim topMatNom As String
    Dim topMatProm As Double
    Dim topPrecioNom As String
    Dim topPrecioImp As Double
    
    PrepararDatosGraficos wsData, wsDatos, lastRow, topCentroNom, topCentroProm, topMatNom, topMatProm, topPrecioNom, topPrecioImp

    ' Cálculo de Data Health Score (Precisión de Carga)
    Dim healthScore As Double
    Dim strHealth As String
    If totalReg > 0 Then
        healthScore = (totalReg - totalAnom) / CDbl(totalReg)
    Else
        healthScore = 1#
    End If
    strHealth = Format(healthScore, "0.0%")

    ' --- 4. CAJA DE RESUMEN DINÁMICO (FILA 4) ---
    wsDash.Range("B4:M4").Interior.Color = CLR_SUMMARY_BG
    Dim resTexto As String
    If totalReg = 0 Then
        resTexto = "  " & ChrW(9888) & "  No se detectaron registros cargados para procesar."
    ElseIf totalAnom = 0 Then
        resTexto = "  " & ChrW(10004) & "  La carga de SAP se procesó con " & strHealth & " de precisión. El centro con mayor importe promedio es " & topCentroNom & " (" & Format(topCentroProm, "$#,##0.00") & ")."
    Else
        resTexto = "  " & ChrW(9888) & "  La carga de SAP se procesó con " & strHealth & " de precisión (" & totalAnom & " registros con anomalías). El centro con mayor importe promedio es " & topCentroNom & " (" & Format(topCentroProm, "$#,##0.00") & ")."
    End If
    DibujarCeldaMerge wsDash, 4, 2, 4, 13, resTexto, 9, False, CLR_TEXT_PRI, xlLeft, xlCenter
    wsDash.Range("B4:M4").Font.Italic = True
    With wsDash.Range("B4:M4").Borders(xlEdgeBottom)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .Color = CLR_CARD_BORDER
    End With
    With wsDash.Range("B4:M4").Borders(xlEdgeTop)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .Color = CLR_CARD_BORDER
    End With

    ' --- 5. TARJETAS KPI (SHAPES FLOTANTES CON SOMBRA Y ACENTO) ---
    wsDash.Range("B5:M9").Interior.Color = RGB(241, 245, 249)
    wsDash.Range("B5:M9").Borders.LineStyle = xlNone

    Dim kpiTitulos(3) As String
    Dim kpiValores(3) As String
    Dim kpiSubs(3) As String
    Dim kpiCols(3) As Integer
    Dim kpiClrVal(3) As Long
    Dim kpiClrAcc(3) As Long
    
    ' KPI 1: Total Registros
    kpiTitulos(0) = ChrW(&HD83D) & ChrW(&HDFCA) & "  TOTAL REGISTROS"
    kpiValores(0) = Format(totalReg, "#,##0")
    kpiSubs(0) = "Consolidado de datos"
    kpiCols(0) = 2
    kpiClrVal(0) = CLR_NAVY
    kpiClrAcc(0) = CLR_NAVY
    
    ' KPI 2: Importe Promedio
    kpiTitulos(1) = ChrW(&HD83D) & ChrW(&HDCB0) & "  IMPORTE PROMEDIO"
    kpiValores(1) = Format(impProm, "$#,##0.00")
    kpiSubs(1) = "Media general"
    kpiCols(1) = 5
    kpiClrVal(1) = CLR_NAVY
    kpiClrAcc(1) = CLR_NAVY
    
    ' KPI 3: Mix Organizaciones

    For tr = 2 To 7

        .LineStyle = xlContinuous
        .Weight = xlMedium
        .Color = CLR_CARD_BORDER
    End With

        DibujarCeldaMerge wsDash, 26, 2, 28, 13, ChrW(10004) & "  Sin anomalías detectadas. Todos los registros cumplen con los criterios de validación.", _
            9, False, CLR_GREEN_TXT, xlCenter, xlCenter
            
        wsDash.Range(wsDash.Cells(26, 2), wsDash.Cells(28, 13)).Interior.Color = CLR_GREEN_BG
        AplicarBordeTarjeta wsDash.Range(wsDash.Cells(26, 2), wsDash.Cells(28, 13)), CLR_GREEN_TXT
    Else
        DibujarCeldaMerge wsDash, 26, 2, 26, 13, "  MUESTREO DE REGISTROS CON ERRORES O IMPORTE CERO", _
                           9, True, CLR_ALERT_TXT, xlLeft, xlCenter
                           
        wsDash.Range(wsDash.Cells(26, 2), wsDash.Cells(26, 13)).Interior.Color = CLR_WHITE
        
        Dim auditHdr(4) As String
        Dim auditCS(4) As Integer
        Dim auditCE(4) As Integer

        auditHdr(1) = "MATERIAL": auditCS(1) = 4: auditCE(1) = 6
        auditHdr(2) = "DENOMINACION": auditCS(2) = 7: auditCE(2) = 9
        auditHdr(3) = "IMPORTE": auditCS(3) = 10: auditCE(3) = 11
        auditHdr(4) = "TIPO ANOMALIA": auditCS(4) = 12: auditCE(4) = 13
        
        Dim ah As Integer
        For ah = 0 To 4
            DibujarCeldaMerge wsDash, 27, auditCS(ah), 27, auditCE(ah), auditHdr(ah), 8, True, CLR_WHITE, xlCenter, xlCenter

        anomRow = 28
        anomCount = 0
        Const MAX_ANOM As Long = 25
        Dim ai As Long
        
        For ai = 0 To totalAnom - 1

                rowBg = CLR_CARD_BG

                rowBg = CLR_ROW_ALT
            End If
            
            wsDash.Rows(anomRow).RowHeight = 15
            
            DibujarCeldaMerge wsDash, anomRow, 2, anomRow, 3, Trim(CStr(wsDatos.Cells(fila, 8).Value)), 8, False, CLR_TEXT_PRI, xlLeft, xlCenter

            wsDash.Range(wsDash.Cells(anomRow, 4), wsDash.Cells(anomRow, 6)).Interior.Color = rowBg
            
            DibujarCeldaMerge wsDash, anomRow, 7, anomRow, 9, Trim(CStr(wsDatos.Cells(fila, 10).Value)), 8, False, CLR_TEXT_SEC, xlLeft, xlCenter

            Dim impA As Double

            If Err.Number <> 0 Then

            On Error GoTo ErrDash

            wsDash.Range(wsDash.Cells(anomRow, 10), wsDash.Cells(anomRow, 11)).Interior.Color = rowBg
            
            Dim tpAnom As String
            Dim impVA As Variant
            impVA = wsDatos.Cells(fila, 11).Value
            
            If Not IsNumeric(impVA) Then
                tpAnom = "IMPORTE INVALIDO"
            ElseIf CDbl(impVA) <= 0 Then

            Else
                tpAnom = "DISCREPANCIA ORG"
            End If
            
            DibujarCeldaMerge wsDash, anomRow, 12, anomRow, 13, tpAnom, 7, True, CLR_ALERT_TXT, xlCenter, xlCenter
            wsDash.Range(wsDash.Cells(anomRow, 12), wsDash.Cells(anomRow, 13)).Interior.Color = CLR_ALERT_BG
            AplicarBordeTarjeta wsDash.Range(wsDash.Cells(anomRow, 12), wsDash.Cells(anomRow, 13)), CLR_ALERT_TXT
            
            With wsDash.Range(wsDash.Cells(anomRow, 2), wsDash.Cells(anomRow, 13)).Borders(xlEdgeBottom)
                .LineStyle = xlContinuous
                .Weight = xlThin

        Next ai

        wsDash.Rows(footRow).RowHeight = 12

                           6, False, CLR_TEXT_SEC, xlCenter, xlCenter
    End If

    ' --- 8. REAFIRMAR SIDEBAR Y BOTÓN DE ACCIÓN ---
    wsDash.Columns("A").Interior.Color = CLR_SIDEBAR

FinDash:

End Sub

Public Function ContarYClasificarAnomalias(wsDatos As Worksheet, lastRow As Long, ByRef anomalias() As Long) As Long
    ReDim anomalias(0 To 5000)
    Dim count As Long
    count = 0
    Dim i As Long
    
    For i = 2 To lastRow
        Dim em As String
        Dim md As String
        Dim impV As Variant
        
        em = Trim(CStr(wsDatos.Cells(i, 5).Value))
        md = Trim(CStr(wsDatos.Cells(i, 3).Value))
        impV = wsDatos.Cells(i, 11).Value
        
        Dim esAnom As Boolean
        esAnom = False
        
        If Not IsNumeric(impV) Then
            esAnom = True
        ElseIf CDbl(impV) <= 0 Then
            esAnom = True
        End If
        
        If Not esAnom Then
            If em = "7100" Then
                If md <> "ZMAH" And md <> "ZMPH" Then
                    esAnom = True
                End If
            ElseIf em = "7180" Then
                If md <> "ZMA6" And md <> "ZMP1" Then
                    esAnom = True
                End If
            End If
        End If
        
        If esAnom Then
            If count > UBound(anomalias) - 1 Then
                ReDim Preserve anomalias(0 To UBound(anomalias) + 1000)
            End If
            anomalias(count) = i
            count = count + 1
        End If
    Next i
    
    ContarYClasificarAnomalias = count
End Function

Public Sub PrepararDatosGraficos(wsData As Worksheet, wsDatos As Worksheet, lastRow As Long, _
                                Optional ByRef topCentroNom As String, _
                                Optional ByRef topCentroProm As Double, _
                                Optional ByRef topMatNom As String, _
                                Optional ByRef topMatProm As Double, _
                                Optional ByRef topPrecioNom As String, _
                                Optional ByRef topPrecioImp As Double)
    wsData.Cells.Clear
    
    ' Encabezados en _DashData
    wsData.Cells(1, 1).Value = "Centro"
    wsData.Cells(1, 2).Value = "Promedio"
    wsData.Cells(1, 4).Value = "Denominacion"
    wsData.Cells(1, 5).Value = "Promedio"
    wsData.Cells(1, 7).Value = "Denominacion"
    wsData.Cells(1, 8).Value = "Importe"
    
    ' 1. Estructuras para Centros (Columna 8 Centro Desc / Columna 11 Importe)
    Dim centros() As String
    Dim sumasCentros() As Double
    Dim conteosCentros() As Long
    Dim nCentros As Long
    nCentros = 0
    ReDim centros(0)
    ReDim sumasCentros(0)
    ReDim conteosCentros(0)
    
    ' 2. Estructuras para Materiales por Promedio (Columna 10 Denominación / Columna 11 Importe)
    Dim materiales() As String
    Dim sumasMat() As Double
    Dim conteosMat() As Long
    Dim nMateriales As Long
    nMateriales = 0
    ReDim materiales(0)
    ReDim sumasMat(0)
    ReDim conteosMat(0)
    
    ' 3. Estructuras para Top Precios Históricos Más Altos (Registros Individuales)
    Dim maxDen() As String
    Dim maxImp() As Double
    Dim nPrecios As Long
    nPrecios = 0
    ReDim maxDen(0 To 1000)
    ReDim maxImp(0 To 1000)
    
    Dim i As Long, j As Long
    Dim cn As String, den As String
    Dim valorImporte As Variant
    Dim found As Boolean
    
    For i = 2 To lastRow
        cn = Trim(CStr(wsDatos.Cells(i, 8).Value))
        den = Trim(CStr(wsDatos.Cells(i, 10).Value))
        valorImporte = wsDatos.Cells(i, 11).Value
        
        If cn = "" Then cn = "(Sin Centro)"
        If den = "" Then
            den = Trim(CStr(wsDatos.Cells(i, 9).Value))
            If den = "" Then den = "(Sin Denominación)"
        End If
        
        ' --- 1. Acumulador para Centros ---
        found = False
        For j = 0 To nCentros - 1
            If centros(j) = cn Then
                If IsNumeric(valorImporte) Then
                    If CDbl(valorImporte) > 0 Then
                        sumasCentros(j) = sumasCentros(j) + CDbl(valorImporte)
                        conteosCentros(j) = conteosCentros(j) + 1
                    End If
                End If
                found = True
                Exit For
            End If
        Next j
        
        If Not found Then
            ReDim Preserve centros(0 To nCentros)
            ReDim Preserve sumasCentros(0 To nCentros)
            ReDim Preserve conteosCentros(0 To nCentros)
            centros(nCentros) = cn
            If IsNumeric(valorImporte) Then
                If CDbl(valorImporte) > 0 Then
                    sumasCentros(nCentros) = CDbl(valorImporte)
                    conteosCentros(nCentros) = 1
                End If
            End If
            nCentros = nCentros + 1
        End If
        
        ' --- 2. Acumulador para Materiales (Agrupado por Denominación Col 10, Importe Col 11) ---
        found = False
        For j = 0 To nMateriales - 1
            If materiales(j) = den Then
                If IsNumeric(valorImporte) Then
                    If CDbl(valorImporte) > 0 Then
                        sumasMat(j) = sumasMat(j) + CDbl(valorImporte)
                        conteosMat(j) = conteosMat(j) + 1
                    End If
                End If
                found = True
                Exit For
            End If
        Next j
        
        If Not found Then
            ReDim Preserve materiales(0 To nMateriales)
            ReDim Preserve sumasMat(0 To nMateriales)
            ReDim Preserve conteosMat(0 To nMateriales)
            materiales(nMateriales) = den
            If IsNumeric(valorImporte) Then
                If CDbl(valorImporte) > 0 Then
                    sumasMat(nMateriales) = CDbl(valorImporte)
                    conteosMat(nMateriales) = 1
                End If
            End If
            nMateriales = nMateriales + 1
        End If
        
        ' --- 3. Registro individual para Precios Históricos Más Altos ---
        If IsNumeric(valorImporte) Then
            If CDbl(valorImporte) > 0 Then
                If nPrecios > UBound(maxImp) Then
                    ReDim Preserve maxDen(0 To UBound(maxDen) + 2000)
                    ReDim Preserve maxImp(0 To UBound(maxImp) + 2000)
                End If
                maxDen(nPrecios) = den
                maxImp(nPrecios) = CDbl(valorImporte)
                nPrecios = nPrecios + 1
            End If
        End If
    Next i
    
    ' --- PROCESAMIENTO CENTROS: PROMEDIOS Y ORDENAMIENTO ---
    Dim promediosCentros() As Double
    Dim tmpD As Double, tmpS As String
    
    If nCentros > 0 Then
        ReDim promediosCentros(0 To nCentros - 1)
        For j = 0 To nCentros - 1
            If conteosCentros(j) > 0 Then
                promediosCentros(j) = sumasCentros(j) / conteosCentros(j)
            Else
                promediosCentros(j) = 0
            End If
        Next j
        
        For i = 0 To nCentros - 2
            For j = 0 To nCentros - 2 - i
                If promediosCentros(j) < promediosCentros(j + 1) Then
                    tmpD = promediosCentros(j)
                    promediosCentros(j) = promediosCentros(j + 1)
                    promediosCentros(j + 1) = tmpD
                    
                    tmpS = centros(j)
                    centros(j) = centros(j + 1)
                    centros(j + 1) = tmpS
                End If
            Next j
        Next i
        
        topCentroNom = centros(0)
        topCentroProm = promediosCentros(0)
    Else
        topCentroNom = "N/A"
        topCentroProm = 0
    End If
    
    Dim top5Centros As Long
    If nCentros >= 5 Then top5Centros = 5 Else top5Centros = nCentros
    
    Dim wrC As Long
    wrC = 2
    For i = top5Centros - 1 To 0 Step -1
        Dim lblC As String
        lblC = centros(i)
        If Len(lblC) > 20 Then lblC = Left(lblC, 18) & "..."
        wsData.Cells(wrC, 1).NumberFormat = "@"
        wsData.Cells(wrC, 1).Value = lblC
        wsData.Cells(wrC, 2).Value = promediosCentros(i)
        wrC = wrC + 1
    Next i
    
    ' --- PROCESAMIENTO MATERIALES (DENOMINACIÓN): PROMEDIOS Y ORDENAMIENTO ---
    Dim promediosMat() As Double
    If nMateriales > 0 Then
        ReDim promediosMat(0 To nMateriales - 1)
        For j = 0 To nMateriales - 1
            If conteosMat(j) > 0 Then
                promediosMat(j) = sumasMat(j) / conteosMat(j)
            Else
                promediosMat(j) = 0
            End If
        Next j
        
        For i = 0 To nMateriales - 2
            For j = 0 To nMateriales - 2 - i
                If promediosMat(j) < promediosMat(j + 1) Then
                    tmpD = promediosMat(j)
                    promediosMat(j) = promediosMat(j + 1)
                    promediosMat(j + 1) = tmpD
                    
                    tmpS = materiales(j)
                    materiales(j) = materiales(j + 1)
                    materiales(j + 1) = tmpS
                End If
            Next j
        Next i
        
        topMatNom = materiales(0)
        topMatProm = promediosMat(0)
    Else
        topMatNom = "N/A"
        topMatProm = 0
    End If
    
    Dim top5Mat As Long
    If nMateriales >= 5 Then top5Mat = 5 Else top5Mat = nMateriales
    
    Dim wrM As Long
    wrM = 2
    For i = top5Mat - 1 To 0 Step -1
        Dim lblM As String
        lblM = materiales(i)
        If Len(lblM) > 20 Then lblM = Left(lblM, 18) & "..."
        wsData.Cells(wrM, 4).NumberFormat = "@"
        wsData.Cells(wrM, 4).Value = lblM
        wsData.Cells(wrM, 5).Value = promediosMat(i)
        wrM = wrM + 1
    Next i
    
    ' --- PROCESAMIENTO PRECIOS HISTÓRICOS MÁS ALTOS (TOP 5 INDIVIDUALES) ---
    Dim top5Precios As Long
    If nPrecios >= 5 Then top5Precios = 5 Else top5Precios = nPrecios
    
    Dim p As Long, q As Long, maxIdx As Long
    For p = 0 To top5Precios - 1
        maxIdx = p
        For q = p + 1 To nPrecios - 1
            If maxImp(q) > maxImp(maxIdx) Then
                maxIdx = q
            End If
        Next q
        If maxIdx <> p Then
            tmpD = maxImp(p): maxImp(p) = maxImp(maxIdx): maxImp(maxIdx) = tmpD
            tmpS = maxDen(p): maxDen(p) = maxDen(maxIdx): maxDen(maxIdx) = tmpS
        End If
    Next p
    
    If top5Precios > 0 Then
        topPrecioNom = maxDen(0)
        topPrecioImp = maxImp(0)
    Else
        topPrecioNom = "N/A"
        topPrecioImp = 0
    End If
    
    Dim wrP As Long
    wrP = 2
    For i = top5Precios - 1 To 0 Step -1
        Dim lblP As String
        lblP = maxDen(i)
        If Len(lblP) > 20 Then lblP = Left(lblP, 18) & "..."
        wsData.Cells(wrP, 7).NumberFormat = "@"
        wsData.Cells(wrP, 7).Value = lblP
        wsData.Cells(wrP, 8).Value = maxImp(i)
        wrP = wrP + 1
    Next i
End Sub

Public Sub EstilarGraficoBarras(cht As Chart, wsData As Worksheet, _
                                colCat As Integer, colVal As Integer, _
                                numRows As Long, titulo As String)
    cht.ChartType = xlBarClustered
    cht.SetSourceData wsData.Range(wsData.Cells(1, colCat), wsData.Cells(numRows + 1, colVal)), xlColumns
    cht.HasTitle = True
    cht.ChartTitle.Text = titulo
    
    With cht.ChartTitle.Font
        .Name = "Segoe UI"
        .Size = 9
        .Bold = True
        .Color = CLR_TEXT_PRI
    End With
    
    cht.HasLegend = False
    
    With cht.PlotArea
        .Interior.Color = CLR_WHITE
    End With
    
    With cht.ChartArea
        .Interior.Color = CLR_CARD_BG
    End With
    
    On Error Resume Next
    cht.ChartArea.Border.LineStyle = xlNone
    cht.PlotArea.Border.LineStyle = xlNone
    
    ' Reducción del grosor de barras aumentando la separación entre categorías
    cht.ChartGroups(1).GapWidth = 150
    
    If cht.SeriesCollection.count > 0 Then
        With cht.SeriesCollection(1)
            .XValues = wsData.Range(wsData.Cells(2, colCat), wsData.Cells(numRows + 1, colCat))
            .Values = wsData.Range(wsData.Cells(2, colVal), wsData.Cells(numRows + 1, colVal))
            .Interior.Color = CLR_NAVY
            .Border.LineStyle = xlNone
            .HasDataLabels = True
            With .DataLabels
                .ShowValue = True
                .NumberFormat = "$#,##0"
                .Font.Name = "Segoe UI"
                .Font.Size = 7
                .Font.Color = CLR_TEXT_PRI
            End With
        End With
    End If
    
    ' Eje vertical de categorías (Denominación / Centro) sin cuadrícula
    With cht.Axes(xlCategory)
        .HasMajorGridlines = False
        .HasMinorGridlines = False
        .TickLabels.Font.Name = "Segoe UI"
        .TickLabels.Font.Size = 7
        .TickLabels.Font.Color = CLR_TEXT_PRI
    End With
    
    ' Eje horizontal de valores sin cuadrícula de fondo y sin etiquetas redundantes
    With cht.Axes(xlValue)
        .HasMajorGridlines = False
        .HasMinorGridlines = False
        .TickLabelPosition = xlTickLabelPositionNone
    End With
    On Error GoTo 0
End Sub

Public Sub DibujarTarjetaKPI(ws As Worksheet, _
                            ByVal idCard As Integer, _
                            ByVal leftPos As Double, ByVal topPos As Double, _
                            ByVal widthPos As Double, ByVal heightPos As Double, _
                            ByVal strTitulo As String, _
                            ByVal strValor As String, _
                            ByVal strSub As String, _
                            ByVal valColor As Long, _
                            ByVal accentColor As Long)
    On Error Resume Next
    ws.Shapes("KpiCard_" & idCard).Delete
    ws.Shapes("KpiAccent_" & idCard).Delete
    On Error GoTo 0
    
    ' 1. Tarjeta base rectangular con fondo claro y sombra sutil
    Dim shpCard As Shape
    Set shpCard = ws.Shapes.AddShape(msoShapeRectangle, leftPos, topPos, widthPos, heightPos)
    shpCard.Name = "KpiCard_" & idCard
    
    With shpCard
        .Fill.Solid
        .Fill.ForeColor.RGB = CLR_CARD_BG
        .Line.Visible = msoFalse
        
        With .Shadow
            .Visible = msoTrue
            .Transparency = 0.8
            .Blur = 6
            .OffsetX = 0
            .OffsetY = 2
            .ForeColor.RGB = RGB(100, 116, 139)
        End With
    End With
    
    ' 2. Línea de acento superior (3.5 px de alto)
    Dim shpAccent As Shape
    Set shpAccent = ws.Shapes.AddShape(msoShapeRectangle, leftPos, topPos, widthPos, 3.5)
    shpAccent.Name = "KpiAccent_" & idCard
    
    With shpAccent
        .Fill.Solid
        .Fill.ForeColor.RGB = accentColor
        .Line.Visible = msoFalse
        .Shadow.Visible = msoFalse
    End With
    
    ' 3. Tipografía y contenido estructurado dentro de la tarjeta
    With shpCard.TextFrame2
        .WordWrap = msoTrue
        .MarginLeft = 4
        .MarginRight = 4
        .MarginTop = 8
        .MarginBottom = 4
        .VerticalAnchor = msoAnchorMiddle
        
        With .TextRange
            .Text = strTitulo & vbCrLf & strValor & vbCrLf & strSub
            .ParagraphFormat.Alignment = msoAlignCenter
            
            ' Párrafo 1: Título con Icono
            With .Paragraphs(1).Font
                .Name = "Segoe UI"
                .Size = 7.5
                .Bold = msoTrue
                .Fill.ForeColor.RGB = CLR_TEXT_SEC
            End With
            
            ' Párrafo 2: Valor Principal
            With .Paragraphs(2).Font
                .Name = "Segoe UI"
                .Size = 18
                .Bold = msoTrue
                .Fill.ForeColor.RGB = valColor
            End With
            
            ' Párrafo 3: Subtítulo descriptivo
            With .Paragraphs(3).Font
                .Name = "Segoe UI"
                .Size = 7
                .Bold = msoFalse
                .Fill.ForeColor.RGB = CLR_TEXT_SEC
            End With
        End With
    End With
End Sub

Public Sub DibujarCeldaMerge(ws As Worksheet, _
                                fila1 As Long, col1 As Integer, _
                                fila2 As Long, col2 As Integer, _
                                texto As String, _
                                sz As Integer, bBold As Boolean, clr As Long, _
                                hAlign As XlHAlign, vAlign As Long)
    Dim rng As Range
    Set rng = ws.Range(ws.Cells(fila1, col1), ws.Cells(fila2, col2))
    
    On Error Resume Next
    rng.UnMerge
    On Error GoTo 0
    
    rng.Merge
    
    With rng
        .Value = texto
        .HorizontalAlignment = hAlign
        .VerticalAlignment = vAlign
        .WrapText = False
    End With
    
    With rng.Font
        .Name = "Segoe UI"
        .Size = sz
        .Bold = bBold
        .Color = clr
    End With
End Sub

Public Sub AplicarBordeTarjeta(rng As Range, clr As Long)
    On Error Resume Next
    With rng.Borders(xlEdgeLeft)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .Color = clr
    End With
    With rng.Borders(xlEdgeTop)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .Color = clr
    End With
    With rng.Borders(xlEdgeRight)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .Color = clr
    End With
    With rng.Borders(xlEdgeBottom)
        .LineStyle = xlContinuous
        .Weight = xlThin
        .Color = clr
    End With
    
    rng.Borders(xlInsideHorizontal).LineStyle = xlNone
    rng.Borders(xlInsideVertical).LineStyle = xlNone
    On Error GoTo 0
End Sub

Public Function ObtenerOCrearHoja(nombre As String) As Worksheet
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(nombre)
    On Error GoTo 0
    
    If ws Is Nothing Then
        Set ws = ThisWorkbook.Worksheets.Add()
        ws.Name = nombre
    End If
    
    Set ObtenerOCrearHoja = ws
End Function

Public Function ObtenerHoja(nombre As String) As Worksheet
    On Error Resume Next
    Set ObtenerHoja = ThisWorkbook.Worksheets(nombre)
    On Error GoTo 0
End Function
